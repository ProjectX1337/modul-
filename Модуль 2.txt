Шаг 2. Импорт данных из Заказчики.json
Здесь есть нюанс: SQL Server не умеет напрямую импортировать JSON через кнопку как Excel. Есть два пути:

Вариант А (простой) — вставить данные вручную через INSERT
Поскольку в файле всего 6 записей, самый быстрый способ — просто вставить их запросом. Выполни в SSMS:

sql

GO

sql

Должно вернуться 6 строк. ✅

Вариант Б (правильный, «по-взрослому») — через OPENJSON
Этот способ использует встроенную функцию SQL Server для работы с JSON. Работает только на SQL Server 2016 и новее.

Сохрани файл Заказчики.json на диск, например: C:\Data\Заказчики.json

Выполни:

sql
USE MilkPlant;
GO

-- Включаем доступ к файлам на диске
EXEC sp_configure 'show advanced options', 1;
RECONFIGURE;
EXEC sp_configure 'Ad Hoc Distributed Queries', 1;
RECONFIGURE;
GO

-- Загружаем JSON из файла
DECLARE @json NVARCHAR(MAX);

SELECT @json = BulkColumn
FROM OPENROWSET (BULK 'C:\Data\Заказчики.json', SINGLE_CLOB) AS j;

-- Разбираем JSON и вставляем в таблицу
INSERT INTO Counterparty (counterparty_id, name, inn, address, phone, is_salesman, is_buyer)
SELECT 
    id,
    name,
    ISNULL(inn, ''),
    addres,
    phone,
    CAST(salesman AS BIT),
    CAST(buyer AS BIT)
FROM OPENJSON(@json)
WITH (
    id        CHAR(9)        '$.id',
    name      NVARCHAR(255)  '$.name',
    inn       NVARCHAR(12)   '$.inn',
    addres    NVARCHAR(255)  '$.addres',
    phone     NVARCHAR(20)   '$.phone',
    salesman  BIT            '$.salesman',
    buyer     BIT            '$.buyer'
);

SELECT * FROM Counterparty;
GO
